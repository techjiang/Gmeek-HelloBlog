// 防止直接访问
if (!defined('ABSPATH')) {
    exit;
}

// 常量定义
define('MY_PLUGIN_VERSION', '1.0.0');
define('MY_PLUGIN_DIR', plugin_dir_path(__FILE__));
define('MY_PLUGIN_URL', plugin_dir_url(__FILE__));

// 激活钩子
register_activation_hook(__FILE__, 'my_plugin_activate');
function my_plugin_activate() {
    // 创建数据库表
    global $wpdb;
    $table_name = $wpdb->prefix . 'my_plugin_data';
    $charset_collate = $wpdb->get_charset_collate();

    $sql = "CREATE TABLE $table_name (
        id mediumint(9) NOT NULL AUTO_INCREMENT,
        name varchar(100) NOT NULL,
        value text,
        created_at datetime DEFAULT CURRENT_TIMESTAMP,
        PRIMARY KEY (id)
    ) $charset_collate;";

    require_once(ABSPATH . 'wp-admin/includes/upgrade.php');
    dbDelta($sql);

    // 设置默认选项
    add_option('my_plugin_settings', array(
        'enabled'      => true,
        'title'        => 'My Plugin',
        'items_per_page' => 10,
    ));

    // 设置默认选项
    flush_rewrite_rules();
}

// 停用钩子
register_deactivation_hook(__FILE__, 'my_plugin_deactivate');
function my_plugin_deactivate() {
    flush_rewrite_rules();
}

// 加载国际化
function my_plugin_load_textdomain() {
    load_plugin_textdomain('my-plugin', false, dirname(plugin_basename(__FILE__)) . '/languages');
}
add_action('init', 'my_plugin_load_textdomain');

// 加载后台脚本
function my_plugin_admin_scripts($hook) {
    if (strpos($hook, 'my-plugin') === false) {
        return;
    }

    wp_enqueue_style('my-plugin-admin', MY_PLUGIN_URL . 'admin/css/admin.css', array(), MY_PLUGIN_VERSION);
    wp_enqueue_script('my-plugin-admin', MY_PLUGIN_URL . 'admin/js/admin.js', array('jquery'), MY_PLUGIN_VERSION, true);

    wp_localize_script('my-plugin-admin', 'myPluginAdmin', array(
        'ajaxUrl' => admin_url('admin-ajax.php'),
        'nonce'   => wp_create_nonce('my_plugin_nonce'),
    ));
}
add_action('admin_enqueue_scripts', 'my_plugin_admin_scripts');

// 加载前台脚本
function my_plugin_frontend_scripts() {
    wp_enqueue_style('my-plugin-frontend', MY_PLUGIN_URL . 'public/css/frontend.css', array(), MY_PLUGIN_VERSION);
    wp_enqueue_script('my-plugin-frontend', MY_PLUGIN_URL . 'public/js/frontend.js', array(), MY_PLUGIN_VERSION, true);
}
add_action('wp_enqueue_scripts', 'my_plugin_frontend_scripts');

9.4 AJAX 处理

<?php
// AJAX 处理（登录用户）
function my_plugin_ajax_handler() {
    // 验证 nonce
    if (!wp_verify_nonce($_POST['nonce'] ?? '', 'my_plugin_nonce')) {
        wp_send_json_error('Security check failed');
    }

    // 验证权限
    if (!current_user_can('manage_options')) {
        wp_send_json_error('Permission denied');
    }

    $action_type = sanitize_text_field($_POST['action_type'] ?? '');

    switch ($action_type) {
        case 'save_data':
            $name = sanitize_text_field($_POST['name'] ?? '');
            $value = sanitize_textarea_field($_POST['value'] ?? '');

            global $wpdb;
            $result = $wpdb->insert($wpdb->prefix . 'my_plugin_data', array(
                'name'  => $name,
                'value' => $value,
            ));

            if ($result) {
                wp_send_json_success(array('message' => 'Data saved successfully'));
            } else {
                wp_send_json_error('Failed to save data');
            }
            break;

        case 'get_data':
            global $wpdb;
            $data = $wpdb->get_results("SELECT * FROM {$wpdb->prefix}my_plugin_data ORDER BY created_at DESC LIMIT 10");
            wp_send_json_success($data);
            break;

        default:
            wp_send_json_error('Unknown action');
    }
}
add_action('wp_ajax_my_plugin_action', 'my_plugin_ajax_handler');

11.2 注册自定义 REST API 端点

<?php
// 注册自定义端点
function my_plugin_register_rest_routes() {
    register_rest_route('my-plugin/v1', '/data', array(
        'methods'             => 'GET',
        'callback'            => 'my_plugin_get_data',
        'permission_callback' => 'my_plugin_check_permission',
        'args'                => array(
            'per_page' => array(
                'default'           => 10,
                'sanitize_callback' => 'absint',
            ),
            'page' => array(
                'default'           => 1,
                'sanitize_callback' => 'absint',
            ),
        ),
    ));

    register_rest_route('my-plugin/v1', '/data', array(
        'methods'             => 'POST',
        'callback'            => 'my_plugin_create_data',
        'permission_callback' => 'my_plugin_check_permission',
    ));

    register_rest_route('my-plugin/v1', '/data/(?P<id>\d+)', array(
        'methods'             => 'PUT',
        'callback'            => 'my_plugin_update_data',
        'permission_callback' => 'my_plugin_check_permission',
    ));

    register_rest_route('my-plugin/v1', '/data/(?P<id>\d+)', array(
        'methods'             => 'DELETE',
        'callback'            => 'my_plugin_delete_data',
        'permission_callback' => 'my_plugin_check_permission',
    ));
}
add_action('rest_api_init', 'my_plugin_register_rest_routes');

// 权限检查
function my_plugin_check_permission($request) {
    return current_user_can('edit_posts');
}

// GET - 获取数据
function my_plugin_get_data($request) {
    global $wpdb;
    $per_page = $request->get_param('per_page');
    $page = $request->get_param('page');
    $offset = ($page - 1) * $per_page;

    $data = $wpdb->get_results($wpdb->prepare(
        "SELECT * FROM {$wpdb->prefix}my_plugin_data LIMIT %d OFFSET %d",
        $per_page, $offset
    ));

    $total = $wpdb->get_var("SELECT COUNT(*) FROM {$wpdb->prefix}my_plugin_data");

    return new WP_REST_Response(array(
        'data'       => $data,
        'total'      => (int) $total,
        'page'       => (int) $page,
        'per_page'   => (int) $per_page,
        'total_pages' => ceil($total / $per_page),
    ), 200);
}

// POST - 创建数据
function my_plugin_create_data($request) {
    global $wpdb;

    $name = sanitize_text_field($request->get_param('name'));
    $value = sanitize_textarea_field($request->get_param('value'));

    $result = $wpdb->insert($wpdb->prefix . 'my_plugin_data', array(
        'name'  => $name,
        'value' => $value,
    ));

    if ($result) {
        return new WP_REST_Response(array(
            'id'      => $wpdb->insert_id,
            'message' => 'Created successfully',
        ), 201);
    }

    return new WP_REST_Response(array('message' => 'Failed to create'), 500);
}

// PUT - 更新数据
function my_plugin_update_data($request) {
    global $wpdb;
    $id = $request->get_param('id');

    $result = $wpdb->update(
        $wpdb->prefix . 'my_plugin_data',
        array(
            'name'  => sanitize_text_field($request->get_param('name')),
            'value' => sanitize_textarea_field($request->get_param('value')),
        ),
        array('id' => $id)
    );

    return new WP_REST_Response(array('message' => 'Updated successfully'), 200);
}

// DELETE - 删除数据
function my_plugin_delete_data($request) {
    global $wpdb;
    $id = $request->get_param('id');

    $result = $wpdb->delete($wpdb->prefix . 'my_plugin_data', array('id' => $id));

    return new WP_REST_Response(array('message' => 'Deleted successfully'), 200);
}

UPDATE wp_options SET option_value = 'https://new-domain.com' WHERE option_name = 'siteurl';
UPDATE wp_options SET option_value = 'https://new-domain.com' WHERE option_name = 'home';

-- 修改管理员密码
UPDATE wp_users SET user_pass = MD5('newpassword') WHERE user_login = 'admin';
-- 注意：WordPress 使用 phpass，建议通过后台修改

-- 禁用所有插件
UPDATE wp_options SET option_value = 'a:0:{}' WHERE option_name = 'active_plugins';

-- 修改主题
UPDATE wp_options SET option_value = 'twentytwentyfour' WHERE option_name = 'template';
UPDATE wp_options SET option_value = 'twentytwentyfour' WHERE option_name = 'stylesheet';

-- 修改固定链接（/post-name/）
UPDATE wp_options SET option_value = '/%postname%/' WHERE option_name = 'permalink_structure';

-- 查看文章和自定义字段
SELECT p.ID, p.post_title, p.post_status, pm.meta_key, pm.meta_value
FROM wp_posts p
LEFT JOIN wp_postmeta pm ON p.ID = pm.post_id
WHERE p.post_type = 'post' AND p.post_status = 'publish'
ORDER BY p.post_date DESC;

-- 批量替换 URL（域名迁移）
UPDATE wp_options SET option_value = REPLACE(option_value, 'http://old.com', 'https://new.com') WHERE option_name = 'siteurl' OR option_name = 'home';
UPDATE wp_posts SET post_content = REPLACE(post_content, 'http://old.com', 'https://new.com');
UPDATE wp_posts SET guid = REPLACE(guid, 'http://old.com', 'https://new.com');
UPDATE wp_postmeta SET meta_value = REPLACE(meta_value, 'http://old.com', 'https://new.com');

20.1 一键部署脚本

#!/bin/bash
# deploy_wordpress.sh
set -e

DOMAIN="example.com"
DB_NAME="wordpress"
DB_USER="wpuser"
DB_PASS=$(openssl rand -base64 24)
WP_DIR="/var/www/wordpress"
ADMIN_USER="admin"
ADMIN_PASS=$(openssl rand -base64 16)
ADMIN_EMAIL="admin@example.com"

echo "=== WordPress 一键部署 ==="

# 1. 安装 LAMP/LNMP
echo "[1/7] 安装环境..."
apt update && apt install -y nginx php-fpm php-mysql php-curl php-gd php-mbstring php-xml php-zip php-intl mysql-server certbot python3-certbot-nginx

# 2. 创建数据库
echo "[2/7] 创建数据库..."
mysql -u root -p -e "
CREATE DATABASE IF NOT EXISTS $DB_NAME DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE USER IF NOT EXISTS '$DB_USER'@'localhost' IDENTIFIED BY '$DB_PASS';
GRANT ALL PRIVILEGES ON $DB_NAME.* TO '$DB_USER'@'localhost';
FLUSH PRIVILEGES;
"

# 3. 下载 WordPress
echo "[3/7] 下载 WordPress..."
cd /var/www
wget -q https://wordpress.org/latest.tar.gz
tar -xzf latest.tar.gz
chown -R www-data:www-data $WP_DIR

# 4. 配置 wp-config.php
echo "[4/7] 配置 WordPress..."
cd $WP_DIR

20.2 自定义短代码插件

<?php
/**
 * Plugin Name: My Shortcodes
 * Description: 自定义短代码集合
 * Version: 1.0.0
 * Author: Your Name
 * Text Domain: my-shortcodes
 */

if (!defined('ABSPATH')) exit;

// 按钮短代码 [my_button url="#" text="点击这里" color="blue" size="large"]
function my_button_shortcode($atts) {
    $atts = shortcode_atts(array(
        'url'   => '#',
        'text'  => 'Click Here',
        'color' => 'blue',
        'size'  => 'medium',
        'target' => '_self',
    ), $atts, 'my_button');

    return sprintf(
        '<a href="%s" class="my-btn my-btn-%s my-btn-%s" target="%s">%s</a>',
        esc_url($atts['url']),
        esc_attr($atts['color']),
        esc_attr($atts['size']),
        esc_attr($atts['target']),
        esc_html($atts['text'])
    );
}
add_shortcode('my_button', 'my_button_shortcode');

// 信息框短代码 [my_alert type="info" title="提示"]内容[/my_alert]
function my_alert_shortcode($atts, $content = null) {
    $atts = shortcode_atts(array(
        'type'  => 'info',
        'title' => '',
    ), $atts, 'my_alert');

    $types = array('info', 'success', 'warning', 'error');
    $type = in_array($atts['type'], $types) ? $atts['type'] : 'info';

    $html = '<div class="my-alert my-alert-' . esc_attr($type) . '">';
    if (!empty($atts['title'])) {
        $html .= '<strong>' . esc_html($atts['title']) . '</strong>';
    }
    $html .= '<p>' . wp_kses_post(do_shortcode($content)) . '</p>';
    $html .= '</div>';

    return $html;
}
add_shortcode('my_alert', 'my_alert_shortcode');

// 文章列表短代码 [my_posts category="news" count="5" layout="grid"]
function my_posts_shortcode($atts) {
    $atts = shortcode_atts(array(
        'category' => '',
        'count'    => 5,
        'layout'   => 'list',
    ), $atts, 'my_posts');

    $args = array(
        'post_type'      => 'post',
        'posts_per_page' => intval($atts['count']),
        'orderby'        => 'date',
        'order'          => 'DESC',
    );

    if (!empty($atts['category'])) {
        $args['category_name'] = sanitize_text_field($atts['category']);
    }

    $query = new WP_Query($args);
    $html = '<div class="my-posts my-posts-' . esc_attr($atts['layout']) . '">';

    if ($query->have_posts()) {
        while ($query->have_posts()) {
            $query->the_post();
            $html .= '<div class="my-post-item">';
            $html .= '<h3><a href="' . get_permalink() . '">' . get_the_title() . '</a></h3>';
            $html .= '<p>' . get_the_excerpt() . '</p>';
            $html .= '<span class="date">' . get_the_date() . '</span>';
            $html .= '</div>';
        }
    } else {
        $html .= '<p>暂无文章</p>';
    }

    $html .= '</div>';
    wp_reset_postdata();

    return $html;
}
add_shortcode('my_posts', 'my_posts_shortcode');

// 加载短代码样式
function my_shortcodes_styles() {
    wp_enqueue_style('my-shortcodes', plugin_dir_url(__FILE__) . 'assets/css/shortcodes.css', array(), '1.0.0');
}
add_action('wp_enqueue_scripts', 'my_shortcodes_styles');
